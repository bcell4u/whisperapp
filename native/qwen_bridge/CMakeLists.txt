cmake_minimum_required(VERSION 3.22)
project(whisperrecorder_qwen_asr_06 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Embed runtime into native probe" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
option(WR_ENABLE_ARM_DOTPROD "Build an ARM64 ASIMDDP-only production ASR decoder" OFF)
if(WR_ENABLE_ARM_DOTPROD)
  # Keep the feature deterministic for the full llama.cpp/ggml build, not only
  # our wrapper. GGML_NATIVE stays OFF to avoid host-dependent cross-compiles.
  add_compile_options(-march=armv8.2-a+dotprod)
  add_compile_definitions(WR_CPU_ARM_DOTPROD=1)
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/CMakeLists.txt")
  message(FATAL_ERROR "Pinned llama.cpp v0.6.0 checkout missing (provided by workflow)")
endif()
add_subdirectory(llama.cpp llama-build EXCLUDE_FROM_ALL)
add_executable(wr-qwen06-decoder qwen_asr_decode.cpp)
target_link_libraries(wr-qwen06-decoder PRIVATE llama)
