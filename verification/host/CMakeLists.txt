cmake_minimum_required(VERSION 3.21)
project(qwen_sliced_reference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED LLAMA_CPP_SOURCE OR NOT EXISTS "${LLAMA_CPP_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Pass -DLLAMA_CPP_SOURCE=/path/to/independently/checked-out/llama.cpp")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_CPP_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/upstream-llama" EXCLUDE_FROM_ALL)
add_executable(qwen-sliced-reference qwen_reference_sliced.cpp)
target_link_libraries(qwen-sliced-reference PRIVATE llama)
